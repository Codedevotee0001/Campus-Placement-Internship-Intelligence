Query 1: Age of all the student Date Function

Calculates each student's current age using TIMESTAMPDIFF with CURDATE(). Powers the 'Age' column in the candidate table.

SELECT 
    first_name,
    last_name,
    dob,
    TIMESTAMPDIFF(YEAR, dob, CURDATE()) AS age
FROM student;
Query 2: List of students who have applied for at least one internship JOIN & IS NOT NULL

Finds all active candidates who submitted at least one internship application (18 students).

SELECT DISTINCT CONCAT(s.first_name, ' ', s.last_name) AS student_name
FROM student s
JOIN application a ON s.student_id = a.student_id
WHERE a.internship_id IS NOT NULL;
Query 3: Show all placements with company name and package offered ORDER BY DESC

Lists all full-time placement openings with company details ordered by package LPA.

SELECT p.placement_id, c.company_name, p.title, p.domain, p.package_lpa, p.location 
FROM placement p 
JOIN company c ON p.company_id = c.company_id 
ORDER BY p.package_lpa DESC;
Query 4: List all internships with stipend > 9000 WHERE Filter

Retrieves high-paying internships offering monthly stipends strictly above ₹9,000 (Powers the Top Stipend card).

SELECT i.title, c.company_name, i.stipend, i.location
FROM internship i
JOIN company c ON i.company_id = c.company_id
WHERE i.stipend > 9000;

Query 6: Students selected for placements Multi-Table JOIN

Identifies all 14 candidates who received confirmed placement offers across visiting recruiters.

SELECT s.first_name, s.last_name, p.title, c.company_name
FROM application a
JOIN student s ON a.student_id = s.student_id
JOIN placement p ON a.placement_id = p.placement_id
JOIN company c ON p.company_id = c.company_id
WHERE a.status = 'Selected';

Query 9: Average placement package offered by each company Recruiter Benchmark

Powers the core Company Average Packages bar chart showing CTCs across all 8 companies.

SELECT c.company_name, ROUND(AVG(p.package_lpa), 2) AS avg_package
FROM placement p
JOIN company c ON p.company_id = c.company_id
GROUP BY c.company_name;
Query 10: Max and min stipend offered by a company MAX & MIN Aggregate

Powers the stipend range comparison matrix showing highest and lowest compensation offered by employers.

SELECT 
    c.company_name,
    MAX(i.stipend) AS max_stipend,
    MIN(i.stipend) AS min_stipend
FROM internship i
JOIN company c ON i.company_id = c.company_id

Query 11: Top 3 Tech Domains by Average Package LIMIT 3 Ranking

Ranks top tech domains: AI/ML (9.25 LPA), Big Data (8.50 LPA), and Cloud/DevOps (8.35 LPA).

SELECT 
    domain,
    COUNT(placement_id) AS total_jobs,
    ROUND(AVG(package_lpa), 2) AS avg_package_lpa
FROM placement
GROUP BY domain
ORDER BY avg_package_lpa DESC
LIMIT 3;
Query 12: Academic Tier Analysis (CGPA Brackets & Selections) CASE Bracket & HAVING

Tiers candidates into Honors (9.0+), Distinction (8.5 - 8.99), and First Class (< 8.5).

SELECT 
    CASE 
        WHEN COALESCE(s.graduation_cgpa, s.cgpa_12) >= 9.0 THEN 'Honors (9.0+ CGPA)'
        WHEN COALESCE(s.graduation_cgpa, s.cgpa_12) >= 8.5 THEN 'Distinction (8.5 - 8.99 CGPA)'
        ELSE 'First Class (< 8.5 CGPA)'
    END AS academic_tier,
    COUNT(DISTINCT s.student_id) AS total_students,
    COUNT(CASE WHEN a.status = 'Selected' THEN 1 END) AS total_selections,
    ROUND(AVG(COALESCE(s.graduation_cgpa, s.cgpa_12)), 2) AS avg_tier_cgpa
FROM student s
LEFT JOIN application a ON s.student_id = a.student_id
GROUP BY academic_tier
HAVING total_selections >= 1
ORDER BY avg_tier_cgpa DESC;
Query 13: Students who did not apply for any internship or placement LEFT JOIN / IS NULL

Query 14: List all students selected at both internship and placement (Dual Offers) HAVING Clause

Identifies top achievers with dual selections: Aman Sharma, Priya Singh, Simran Kaur, and Sahil Jain.

SELECT 
    s.student_id,
    CONCAT(s.first_name, ' ', s.last_name) AS student_name,
    s.department,
    s.specialization,
    COUNT(CASE WHEN a.internship_id IS NOT NULL AND a.status = 'Selected' THEN 1 END) AS internships_selected,
    COUNT(CASE WHEN a.placement_id IS NOT NULL AND a.status = 'Selected' THEN 1 END) AS placements_selected
FROM student s
JOIN application a ON s.student_id = a.student_id
GROUP BY s.student_id, student_name, s.department, s.specialization
HAVING internships_selected >= 1 AND placements_selected >= 1
ORDER BY s.student_id ASC;
Query 15: Spread salary (Max - Min Package per Company) Salary Variance

Measures salary range within each recruiter (DataSphere has highest spread of 1.70 LPA).

SELECT 
    c.company_name,
    COUNT(p.placement_id) AS total_roles_offered,
    MAX(p.package_lpa) AS highest_package_lpa,
    MIN(p.package_lpa) AS lowest_package_lpa,
    ROUND(MAX(p.package_lpa) - MIN(p.package_lpa), 2) AS package_spread_lpa
FROM company c
JOIN placement p ON c.company_id = p.company_id
GROUP BY c.company_id, c.company_name
HAVING COUNT(p.placement_id) > 1
ORDER BY package_spread_lpa DESC;
Query 16: Above Average Recruiters Correlated Subquery

Filters companies whose average package exceeds university benchmark 7.48 LPA (AIgen, CloudVerse, CyberFort, DataSphere).

SELECT 
    c.company_name,
    c.industry_type,
    c.location,
    ROUND(AVG(p.package_lpa), 2) AS company_avg_package,
    (SELECT ROUND(AVG(package_lpa), 2) FROM placement) AS university_avg_package
FROM company c
JOIN placement p ON c.company_id = p.company_id
GROUP BY c.company_id, c.company_name, c.industry_type, c.location
HAVING AVG(p.package_lpa) > (SELECT AVG(package_lpa) FROM placement)
ORDER BY company_avg_package DESC;
